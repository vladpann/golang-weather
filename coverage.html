type loggerContextKey struct{}

var (
        key = loggerContextKey{}
)

type Logger struct {
        *zap.Logger

        file *os.File
}

func ToContext(ctx context.Context, log *Logger) context.Context {
        return context.WithValue(
                ctx,
                key,
                log,
        )
}

func FromContext(ctx context.Context) *Logger {
        log, ok := ctx.Value(key).(*Logger)
        if !ok {
                panic("no logger in context")
        }

        return log
}

func NewLogger(config Config) (*Logger, error) {
        zapLvl := zap.NewAtomicLevel()
        if err := zapLvl.UnmarshalText([]byte(config.Level)); err != nil {
                return nil, fmt.Errorf("unmarshal log level: %w", err)
        }

        if err := os.MkdirAll(config.Folder, 0755); err != nil {
                return nil, fmt.Errorf("mkfir log folder: %w", err)
        }

        timestamp := time.Now().UTC().Format("2006-01-02T15-04-05.000000")
        logFilePath := filepath.Join(
                config.Folder,
                fmt.Sprintf("%s.log", timestamp),
        )

        logFile, err := os.OpenFile(logFilePath, os.O_CREATE|os.O_WRONLY, 0644)
        if err != nil {
                return nil, fmt.Errorf("open log file: %w", err)
        }

        zapConfig := zap.NewDevelopmentEncoderConfig()
        zapConfig.EncodeTime = zapcore.TimeEncoderOfLayout("2006-01-02T15:04:05.000000")

        zapEncoder := zapcore.NewConsoleEncoder(zapConfig)

        core := zapcore.NewTee(
                zapcore.NewCore(zapEncoder, zapcore.AddSync(os.Stdout), zapLvl),
                zapcore.NewCore(zapEncoder, zapcore.AddSync(logFile), zapLvl),
        )

        zapLogger := zap.New(core, zap.AddCaller())

        return &Logger{
                Logger: zapLogger,
                file:   logFile,
        }, nil
}

func (l *Logger) With(field ...zap.Field) *Logger {
        return &Logger{
                Logger: l.Logger.With(field...),
                file:   l.file,
        }
}

func (l *Logger) Close() {
        if err := l.file.Close(); err != nil {
                fmt.Println("failed to close application logger: ", err)
        }
}

func mapErrors(err error) error {
        const (
                pgxViolatesForeignKeyErrorCode = "23503"
                pgxUniqueViolationErrorCode    = "23505"
        )

        if errors.Is(err, pgx.ErrNoRows) {
                return core_postgres_pool.ErrNoRows

        }

        var pgErr *pgconn.PgError
        if errors.As(err, &pgErr) {
                switch pgErr.Code {
                case pgxViolatesForeignKeyErrorCode:
                        return fmt.Errorf(
                                "%v: %w",
                                err,
                                core_postgres_pool.ErrViolatesForeignKey,
                        )

                case pgxUniqueViolationErrorCode:
                        return fmt.Errorf(
                                "%v: %w",
                                err,
                                core_postgres_pool.ErrUniqueViolation,
                        )
                }
        }

        return fmt.Errorf(
                "%v: %w",
                err,
                core_postgres_pool.ErrUnknown,
        )
}

type contextKey string

const (
        UserIDContextKey    contextKey = "userID"
        SessionIDContextKey contextKey = "sessionID"
)

type SessionsService interface {
        GetSession(
                ctx context.Context,
                sessionID uuid.UUID,
        ) (domain.Session, error)
}

func Auth(sessionsService SessionsService) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        ctx := r.Context()
                        log := core_logger.FromContext(ctx)

                        cookie, err := r.Cookie("session_id")
                        if err != nil {
                                log.Debug("no session cookie found")
                                http.Error(w, "Unauthorized", http.StatusUnauthorized)
                                return
                        }

                        sessionID, err := uuid.Parse(cookie.Value)
                        if err != nil {
                                log.Debug("invalid session ID format")
                                http.Error(w, "Invalid session", http.StatusUnauthorized)
                                return
                        }

                        session, err := sessionsService.GetSession(ctx, sessionID)
                        if err != nil {
                                log.Debug("session not found or expired", zap.Error(err))
                                http.Error(w, "Session not found or expired", http.StatusUnauthorized)
                                return
                        }

                        ctx = context.WithValue(ctx, UserIDContextKey, session.UserID)
                        ctx = context.WithValue(ctx, SessionIDContextKey, sessionID)

                        next.ServeHTTP(w, r.WithContext(ctx))
                })
        }
}

func GetUserIDFromContext(ctx context.Context) (int64, bool) {
        userID, ok := ctx.Value(UserIDContextKey).(int64)
        return userID, ok
}

func GetSessionIDFromContext(ctx context.Context) (uuid.UUID, bool) {
        sessionID, ok := ctx.Value(SessionIDContextKey).(uuid.UUID)
        return sessionID, ok
}

func CORS(allowedOriginsList []string) Middleware {
        allowedOrigins := make(map[string]struct{})
        for _, origin := range allowedOriginsList {
                allowedOrigins[origin] = struct{}{}
        }

        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        origin := r.Header.Get("Origin")

                        if _, ok := allowedOrigins[origin]; ok {
                                w.Header().Set("Access-Control-Allow-Origin", origin)
                                w.Header().Set("Access-Control-Allow-Methods", "GET, POST, PATCH, DELETE, OPTIONS")
                                w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
                                w.Header().Set("Access-Control-Allow-Credentials", "true")
                        }

                        if r.Method == http.MethodOptions {
                                w.WriteHeader(http.StatusOK)
                                return
                        }

                        next.ServeHTTP(w, r)
                })
        }
}

func RequestID() Middleware {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        requestID := r.Header.Get(requestIDHeader)
                        if requestID == "" {
                                requestID = uuid.NewString()
                        }

                        r.Header.Set(requestIDHeader, requestID)
                        w.Header().Set(requestIDHeader, requestID)

                        next.ServeHTTP(w, r)
                })
        }
}

func newResponseWriter(w http.ResponseWriter) *responseWriter {

var sensitivePathPrefixes = []string{
        "/api/v1/auth",
}

func isSensitivePath(path string) bool {
        for _, p := range sensitivePathPrefixes {
                if path == p || strings.HasPrefix(path, p+"/") {
                        return true
                }
        }
        return false
}

func Logger(logger *core_logger.Logger) Middleware {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()

                        requestID := r.Header.Get(requestIDHeader)

                        l := logger.With(
                                zap.String("request_id", requestID),
                                zap.String("url", r.URL.String()),
                        )

                        ctx := core_logger.ToContext(r.Context(), l)

                        sensitive := isSensitivePath(r.URL.Path)

                        var requestBody string
                        if !sensitive && r.Body != nil {
                                bodyBytes, err := io.ReadAll(r.Body)
                                if err == nil {
                                        requestBody = string(bodyBytes)
                                        r.Body = io.NopCloser(bytes.NewBuffer(bodyBytes))
                                }
                        }

                        wrappedWriter := newResponseWriter(w)

                        next.ServeHTTP(wrappedWriter, r.WithContext(ctx))

                        duration := time.Since(start)
                        durationMs := float64(duration.Microseconds()) / 100.0

                        reqField := r.Method + " " + r.RequestURI
                        respField := wrappedWriter.body.String()

                        if sensitive {
                                reqField += " Body: [REDACTED]"
                                respField = "[REDACTED]"
                        }

                        if requestBody != "" {
                                reqField += " Body: " + requestBody
                        }

                        l.Info("HTTP Request",
                                zap.String("time", start.Format(time.RFC3339)),
                                zap.String("req", reqField),
                                zap.String("resp", respField),
                                zap.Int("code", wrappedWriter.statusCode),
                                zap.Float64("duration_ms", durationMs),
                        )
                })
        }
}

type Middleware func(http.Handler) http.Handler

func ChainMiddleware(
        h http.Handler,
        m ...Middleware,
) http.Handler {
        if len(m) == 0 {
                return h
        }

        for i := len(m) - 1; i >= 0; i-- {
                h = m[i](h)
        }

        return h
}

}

func DecodeAndValidateRequest(r *http.Request, dest any) error {
        if err := json.NewDecoder(r.Body).Decode(&dest); err != nil {
                return fmt.Errorf(
                        "decode json: %v: %w",
                        err,
                        core_errors.ErrInvalidArgument,
                )
        }

        var (
                err error
        )

        v, ok := dest.(validatable)
        if ok {
                err = v.Validate()
        } else {
                err = requestValidator.Struct(dest)
        }

        if err != nil {
                return fmt.Errorf(
                        "request validation: %v: %w",
                        err,
                        core_errors.ErrInvalidArgument,
                )
        }

        return nil
}

func (h *HTTPResponseHandler) JSONResponse(responseBody any, statusCode int) {
        h.rw.Header().Set("Content-Type", "application/json")
        h.rw.WriteHeader(statusCode)

        if err := json.NewEncoder(h.rw).Encode(responseBody); err != nil {
                h.log.Error("write HTTP response", zap.Error(err))
        }
}

func (h *HTTPResponseHandler) NoContentResponse() {
        h.rw.WriteHeader(http.StatusNoContent)
}

func (h *HTTPResponseHandler) HTMLResponse(html []byte) {
        h.rw.Header().Set("Content-Type", "text/html; charset=utf-8")
        h.rw.WriteHeader(http.StatusOK)

        if _, err := h.rw.Write(html); err != nil {
                h.log.Error("write HTML HTTP response", zap.Error(err))
        }
}

func (h *HTTPResponseHandler) ErrorResponse(err error, msg string) {
        var (
                statusCode int
                logFunc    func(string, ...zap.Field)
                publicErr  string
        )

        switch {
        case errors.Is(err, core_errors.ErrInvalidArgument):
                statusCode, logFunc, publicErr = http.StatusBadRequest, h.log.Warn, core_errors.ErrInvalidArgument.Error()

        case errors.Is(err, core_errors.ErrNotFound):
                statusCode, logFunc, publicErr = http.StatusNotFound, h.log.Debug, core_errors.ErrNotFound.Error()

        case errors.Is(err, core_errors.ErrConflict):
                statusCode, logFunc, publicErr = http.StatusConflict, h.log.Warn, core_errors.ErrConflict.Error()

        case errors.Is(err, core_errors.ErrLoginAlreadyExists):
                statusCode, logFunc, publicErr = http.StatusConflict, h.log.Warn, core_errors.ErrLoginAlreadyExists.Error()

        case errors.Is(err, core_errors.ErrLocationAlreadyExists):
                statusCode, logFunc, publicErr = http.StatusConflict, h.log.Warn, core_errors.ErrLocationAlreadyExists.Error()

        case errors.Is(err, core_errors.ErrPasswordDoNotMatch):
                statusCode, logFunc, publicErr = http.StatusBadRequest, h.log.Warn, core_errors.ErrPasswordDoNotMatch.Error()

        case errors.Is(err, core_errors.ErrInvalidCredentials):
                statusCode, logFunc, publicErr = http.StatusUnauthorized, h.log.Warn, core_errors.ErrInvalidCredentials.Error()

        default:
                statusCode, logFunc, publicErr = http.StatusInternalServerError, h.log.Error, ""
        }

        logFunc(msg, zap.Error(err))

        h.errorResponse(statusCode, publicErr, msg)
}

func (h *HTTPResponseHandler) PanicResponse(p any, msg string) {
        h.log.Error(
                msg,
                zap.String("panic", fmt.Sprint(p)),
                zap.ByteString("stack", debug.Stack()),
        )

        h.errorResponse(http.StatusInternalServerError, "", msg)
}

func (h *HTTPResponseHandler) errorResponse(
        statusCode int,
        publicErr string,
        msg string,
) {
        h.JSONResponse(
                ErrorResponse{
                        Error:   publicErr,
                        Message: msg,
                },
                statusCode,
        )
}

import (
        "context"
        "errors"
        "fmt"
        "net/http"

        "github.com/go-chi/chi/v5"
        "go.uber.org/zap"

        core_logger "github.com/vladpann/backend/golang-weather/internal/core/logger"
        core_http_middleware "github.com/vladpann/backend/golang-weather/internal/core/transport/http/middleware"
)

type HTTPServer struct {
        router     chi.Router
        config     Config
        log        *core_logger.Logger
        middleware []core_http_middleware.Middleware
}

func NewHTTPServer(
        config Config,
        log *core_logger.Logger,
        middleware ...core_http_middleware.Middleware,
) *HTTPServer {
        r := chi.NewRouter()

        for _, mw := range middleware {
                r.Use(mw)
        }

        return &HTTPServer{
                router:     r,
                config:     config,
                log:        log,
                middleware: middleware,
        }
}

func (s *HTTPServer) RegisterAPIRouters(routers ...*APIVersionRouter) {
        for _, router := range routers {
                prefix := "/api/" + string(router.apiVersion)
                s.router.Mount(prefix, router.Router)
        }
}

func (s *HTTPServer) Run(ctx context.Context) error {
        server := &http.Server{
                Addr:    s.config.Addr,
                Handler: s.router,
        }

        ch := make(chan error, 1)

        go func() {
                defer close(ch)

                s.log.Warn("start HTTP server", zap.String("addr", s.config.Addr))

                err := server.ListenAndServe()

                if !errors.Is(err, http.ErrServerClosed) {
                        ch <- err
                }
        }()

        select {
        case err := <-ch:
                if err != nil {
                        return fmt.Errorf("listen and serve HTTP: %w", err)
                }
        case <-ctx.Done():
                s.log.Warn("shutdown HTTP server...")

                shutdownCtx, cancel := context.WithTimeout(
                        context.Background(),
                        s.config.ShutdownTimeout,
                )
                defer cancel()

                if err := server.Shutdown(shutdownCtx); err != nil {
                        _ = server.Close()

                        return fmt.Errorf("shutdown HTTP server: %w", err)
                }

                s.log.Warn("HTTP server stopped")
        }

        return nil
}

                RETURNING id, user_id, name, latitude, longitude
        `

                        )
                }

                return domain.Location{}, fmt.Errorf("scan location: %w", err)
        }

        locationDomain := locationDomainFromModel(locationModel)

        return locationDomain, nil
}

func (r *LocationsRepository) GetLocations(
        ctx context.Context,
        userID int64,
        limit *int,
        offset *int,
) ([]domain.Location, error) {
        ctx, cancel := context.WithTimeout(ctx, r.pool.OpTimeout())
        defer cancel()

        query := `
        SELECT id, user_id, name, latitude, longitude
        FROM weather.locations
        WHERE user_id = $1
        ORDER BY id ASC
        LIMIT $2
        OFFSET $3;
        `

        rows, err := r.pool.Query(
                ctx,
                query,
                userID,
                limit,
                offset,
        )
        if err != nil {
                return nil, fmt.Errorf("select locations: %w", err)
        }
        defer rows.Close()

        var locationModels []LocationModel
        for rows.Next() {
                var locationModel LocationModel

                err := rows.Scan(
                        &locationModel.ID,
                        &locationModel.UserID,
                        &locationModel.Name,
                        &locationModel.Latitude,
                        &locationModel.Longitude,
                )
                if err != nil {
                        return nil, fmt.Errorf("scan locations: %w", err)
                }

                locationModels = append(locationModels, locationModel)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("next rows: %w", err)
        }

        locationDomains := locationDomainsFromModels(locationModels)

        return locationDomains, nil
}

package ratelimiter

import (
        "context"
        "sync"
        "time"

        core_logger "github.com/vladpann/backend/golang-weather/internal/core/logger"
        "go.uber.org/zap"
)

// queuedRequest represents a request waiting in the queue
type queuedRequest struct {
        ctx       context.Context
        allowChan chan struct{}
        errChan   chan error
}

// RateLimiter manages API request rate limiting using token bucket algorithm
type RateLimiter struct {
        mu             sync.Mutex
        requestsPerMin int
        windowDuration time.Duration
        tokens         int
        lastRefill     time.Time
        queue          chan *queuedRequest
        queueSize      int
        logger         *core_logger.Logger
        stopChan       chan struct{}
        wg             sync.WaitGroup
        closed         bool // Track if already closed
}

// NewRateLimiter creates a new rate limiter instance
func NewRateLimiter(config Config, logger *core_logger.Logger) (*RateLimiter, error) {
        if err := config.Validate(); err != nil {
                return nil, err
        }

        rl := &RateLimiter{
                requestsPerMin: config.RequestsPerMinute,
                windowDuration: time.Minute,
                tokens:         config.RequestsPerMinute,
                lastRefill:     time.Now(),
                queue:          make(chan *queuedRequest, config.QueueSize),
                queueSize:      config.QueueSize,
                logger:         logger,
                stopChan:       make(chan struct{}),
        }

        // Start background queue processor
        rl.wg.Add(1)
        go rl.processQueue()

        logger.Info("rate limiter initialized",
                zap.Int("requests_per_minute", config.RequestsPerMinute),
                zap.Int("queue_size", config.QueueSize))

        return rl, nil
}

// Allow blocks until the request can proceed or context is cancelled
func (rl *RateLimiter) Allow(ctx context.Context) error {
        if err := ctx.Err(); err != nil {
                return err
        }

        // Try to acquire token immediately
        if rl.tryAcquireToken() {
                rl.logger.Debug("token acquired immediately")
                return nil
        }

        // Token not available, queue the request
        qr := &queuedRequest{
                ctx:       ctx,
                allowChan: make(chan struct{}),
                errChan:   make(chan error, 1),
        }

        // Try to add to queue
        select {
        case rl.queue <- qr:
                rl.logger.Debug("request queued", zap.Int("queue_size", len(rl.queue)))
        case <-ctx.Done():
                return ctx.Err()
        default:
                rl.logger.Warn("rate limit queue is full")
                return ErrQueueFull
        }

        // Wait for either allow signal or context cancellation
        select {
        case <-qr.allowChan:
                rl.logger.Debug("queued request allowed")
                return nil
        case err := <-qr.errChan:
                return err
        case <-ctx.Done():
                rl.logger.Warn("request timed out while waiting in queue")
                return ctx.Err()
        }
}

// tryAcquireToken attempts to acquire a token immediately
func (rl *RateLimiter) tryAcquireToken() bool {
        rl.mu.Lock()
        defer rl.mu.Unlock()

        rl.refillTokens()

        if rl.tokens > 0 {
                rl.tokens--
                rl.logger.Debug("token acquired", zap.Int("remaining_tokens", rl.tokens))
                return true
        }

        return false
}

// refillTokens updates available tokens based on elapsed time
func (rl *RateLimiter) refillTokens() {
        now := time.Now()
        elapsed := now.Sub(rl.lastRefill)

        // If a full window has passed, reset to full capacity
        if elapsed >= rl.windowDuration {
                oldTokens := rl.tokens
                rl.tokens = rl.requestsPerMin
                rl.lastRefill = now
                rl.logger.Debug("token bucket refilled",
                        zap.Int("old_tokens", oldTokens),
                        zap.Int("new_tokens", rl.tokens),
                        zap.Duration("elapsed", elapsed))
                return
        }

        // Otherwise, add tokens proportionally to elapsed time
        tokensToAdd := int(float64(rl.requestsPerMin) * elapsed.Seconds() / rl.windowDuration.Seconds())
        if tokensToAdd > 0 {
                oldTokens := rl.tokens
                rl.tokens = min(rl.tokens+tokensToAdd, rl.requestsPerMin)
                rl.lastRefill = now
                rl.logger.Debug("tokens added",
                        zap.Int("old_tokens", oldTokens),
                        zap.Int("new_tokens", rl.tokens),
                        zap.Int("added", tokensToAdd))
        }
}

// processQueue handles queued requests as tokens become available
func (rl *RateLimiter) processQueue() {
        defer rl.wg.Done()

        ticker := time.NewTicker(100 * time.Millisecond)
        defer ticker.Stop()

        for {
                select {
                case <-rl.stopChan:
                        return
                case <-ticker.C:
                        rl.processQueuedRequests()
                }
        }
}

// processQueuedRequests attempts to process pending requests
func (rl *RateLimiter) processQueuedRequests() {
        for {
                // Check if there are queued requests
                if len(rl.queue) == 0 {
                        return
                }

                // Try to acquire a token
                if !rl.tryAcquireToken() {
                        return
                }

                // Get next request from queue
                select {
                case qr := <-rl.queue:
                        // Check if context is still valid
                        select {
                        case <-qr.ctx.Done():
                                rl.logger.Warn("queued request context expired, skipping")
                                // Return token since we didn't use it
                                rl.returnToken()
                                continue
                        default:
                                // Context is valid, allow the request
                                close(qr.allowChan)
                        }
                default:
                        // No request available, return token
                        rl.returnToken()
                        return
                }
        }
}

// returnToken returns a token to the bucket (called when a queued request is cancelled)
func (rl *RateLimiter) returnToken() {
        rl.mu.Lock()
        defer rl.mu.Unlock()

        if rl.tokens < rl.requestsPerMin {
                rl.tokens++
                rl.logger.Debug("token returned", zap.Int("remaining_tokens", rl.tokens))
        }
}

// IsClosed returns whether the rate limiter has been closed
func (rl *RateLimiter) IsClosed() bool {
        rl.mu.Lock()
        defer rl.mu.Unlock()
        return rl.closed
}

// Close stops the rate limiter and cleans up resources
func (rl *RateLimiter) Close() {
        rl.mu.Lock()
        if rl.closed {
                rl.mu.Unlock()
                return // Already closed
        }
        rl.closed = true
        rl.mu.Unlock()

        close(rl.stopChan)
        rl.wg.Wait()
        close(rl.queue)
}

                return

                responseHandler.ErrorResponse(sanitizeRequestError(err), "failed to get open weather location")

        if err := json.NewDecoder(resp.Body).Decode(&weatherData); err != nil {
                responseHandler.ErrorResponse(err, "failed to decode weather response")
                return
        }

        responseHandler.JSONResponse(weatherData, http.StatusOK)
}

// handleRateLimitError converts rate limiter errors to appropriate HTTP responses
func (h *OpenWeatherHTTPHandler) handleRateLimitError(
        responseHandler *core_http_response.HTTPResponseHandler,
        err error,
) {
        switch {
        case errors.Is(err, ratelimiter.ErrQueueFull):
                responseHandler.JSONResponse(
                        map[string]interface{}{
                                "error": "Service temporarily unavailable",
                                "message": fmt.Sprintf(
                                        "Rate limit queue is full (max queue size: %d). Please try again in a few seconds.",
                                        h.config.QueueSize,
                                ),
                                "rate_limit": map[string]interface{}{
                                        "requests_per_minute": h.config.RequestsPerMinute,
                                        "retry_after_seconds": 5,
                                },
                        },
                        http.StatusServiceUnavailable,
                )

        case errors.Is(err, context.DeadlineExceeded):
                responseHandler.JSONResponse(
                        map[string]interface{}{
                                "error":   "Request timeout",
                                "message": "Request timed out while waiting for rate limit. Please try again.",
                                "rate_limit": map[string]interface{}{
                                        "requests_per_minute": h.config.RequestsPerMinute,
                                        "retry_after_seconds": 5,
                                },
                        },
                        http.StatusRequestTimeout,
                )

        case errors.Is(err, context.Canceled):
                responseHandler.JSONResponse(
                        map[string]interface{}{
                                "error":   "Request canceled",
                                "message": "Request was canceled",
                        },
                        http.StatusRequestTimeout,
                )

        default:
                // Generic rate limit error
                responseHandler.JSONResponse(
                        map[string]interface{}{
                                "error": "Rate limit exceeded",
                                "message": fmt.Sprintf(
                                        "Too many requests. Rate limit: %d requests per minute. Please try again in a few seconds.",
                                        h.config.RequestsPerMinute,
                                ),
                                "rate_limit": map[string]interface{}{
                                        "requests_per_minute": h.config.RequestsPerMinute,
                                        "retry_after_seconds": 5,
                                },
                        },
                        http.StatusTooManyRequests,
                )
        }
}

func (p *Hasher) Hash(password string) ([]byte, error) {
        passwordHash, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
        if err != nil {
                return nil, err
        }

        return passwordHash, nil
}

func (p *Hasher) Compare(hash []byte, password string) error {
        err := bcrypt.CompareHashAndPassword(hash, []byte(password))
        if err != nil {
                if errors.Is(err, bcrypt.ErrMismatchedHashAndPassword) {
                        return core_errors.ErrInvalidCredentials
                }
                return fmt.Errorf("compare password: %w", err)
        }

        return nil
}

                return
        }

        scripts, err := filepath.Glob(filepath.Join(migrationsDir, "*.up.sql"))
        if err != nil {
                pgErr = err
                return
        }
        sort.Strings(scripts)

        container, err := postgres.Run(ctx, pgImage,
                postgres.WithDatabase(pgDBName),
                postgres.WithUsername(pgUser),
                postgres.WithPassword(pgPassword),
                postgres.WithInitScripts(scripts...),
                postgres.BasicWaitStrategies(),
        )
        if err != nil {
                pgErr = fmt.Errorf("start postgres container: %w", err)
                return
        }

        if pgHost, err = container.Host(ctx); err != nil {
                pgErr = err
                return
        }

        port, err := container.MappedPort(ctx, "5432/tcp")
        if err != nil {
                pgErr = err
                return
        }
        pgPort = port.Port()
}
